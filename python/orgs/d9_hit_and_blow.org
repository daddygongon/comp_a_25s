#+begin_src python
# one-digit code

# make random number
import random
a = random.randint(0,9)
print(a)

# endless loop
while True:
    # check input
    b_tmp=input("type number:>")
    b = int(b_tmp)
    print(repr(b)) # representation

    # check hit and blow
    hit = 0
    if a==b:
        hit = 4
        print("hit")
    else:
        print("missed")
    
    # ending process
    if hit == 4:
        break
#+end_src
